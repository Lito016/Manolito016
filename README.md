```javascript
class AIAgent {
  constructor(config) {
    this.name = config.name;
    this.tools = config.tools || [];
    this.memory = [];
  }

  async think(task) {
    const context = this.buildContext(task);
    const plan = await this.plan(context);
    return this.execute(plan);
  }

  buildContext(task) {
    return {
      task,
      history: this.memory.slice(-10),
      availableTools: this.tools.map(t => t.schema),
    };
  }

  async execute(plan) {
    const results = [];
    for (const step of plan) {
      const tool = this.tools.find(t => t.name === step.tool);
      const result = await tool.run(step.params);
      results.push({ step, result });
      this.memory.push({ step, result });
    }
    return results;
  }
}
```
